  <script>
    // 기본 내장 N1 고빈출 단어 샘플 (JSON 업로드 시 덮어씌워짐)
    const DEFAULT_N1_VOCAB = [
      { kanji: "躊躇", hiragana: "ちゅうちょ", meaning: "망설임, 주저" },
      { kanji: "緻密", hiragana: "ちみつ", meaning: "치밀함" },
      { kanji: "齟齬", hiragana: "そご", meaning: "어긋남, 서어" },
      { kanji: "牽引", hiragana: "けんいん", meaning: "견인, 이끎" },
      { kanji: "迅速", hiragana: "じんそく", meaning: "신속" },
      { kanji: "隠蔽", hiragana: "いんぺい", meaning: "은폐" },
      { kanji: "軋轢", hiragana: "あつれき", meaning: "알력, 불화" },
      { kanji: "示唆", hiragana: "しさ", meaning: "시사, 암시" },
      { kanji: "頒布", hiragana: "はんぷ", meaning: "반포, 나누어 줌" },
      { kanji: "網羅", hiragana: "もうら", meaning: "망라, 모두 포함함" },
      { kanji: "招聘", hiragana: "しょうへい", meaning: "초빙" },
      { kanji: "補填", hiragana: "ほてん", meaning: "보전, 메움" },
      { kanji: "破綻", hiragana: "はたん", meaning: "파탄" },
      { kanji: "露呈", hiragana: "ろてい", meaning: "드러남, 노정" },
      { kanji: "脆弱", hiragana: "ぜいじゃく", meaning: "취약" },
      { kanji: "乖離", hiragana: "かいり", meaning: "괴리" },
      { kanji: "払拭", hiragana: "ふっしょく", meaning: "불식, 씻어냄" },
      { kanji: "捏造", hiragana: "ねつぞう", meaning: "날조" },
      { kanji: "逡巡", hiragana: "しゅんじゅん", meaning: "주저함, 준순" },
      { kanji: "糾弾", hiragana: "きゅうだん", meaning: "규탄" }
    ];

    const WORDS_PER_DAY = 50; // 하루 50개 고정
    let currentDay = 1;
    let currentCards = [];
    let currentIndex = 0;
    let isFlipped = false;
    let isReviewOnlyMode = false;

    // N1 단어 데이터베이스 로드
    function getN1Database() {
      const stored = localStorage.getItem('N1_VOCAB_DATA');
      return stored ? JSON.parse(stored) : DEFAULT_N1_VOCAB;
    }

    // Day별 단어 Slice (중복 0% 보장)
    function getDayWords(dayNum) {
      const db = getN1Database();
      const startIndex = (dayNum - 1) * WORDS_PER_DAY;
      const daySlice = db.slice(startIndex, startIndex + WORDS_PER_DAY);

      return daySlice.map((item, idx) => ({
        id: `n1_d${dayNum}_w${idx}`,
        kanji: item.kanji,
        hiragana: item.hiragana,
        meaning: item.meaning
      }));
    }

    // Day 그리드 출력
    function renderDayList() {
      const grid = document.getElementById('day-grid');
      grid.innerHTML = '';
      const db = getN1Database();
      const totalWords = db.length;
      const totalDays = Math.max(1, Math.ceil(totalWords / WORDS_PER_DAY));

      document.getElementById('summary-header').innerText = `총 ${totalWords}개 단어 (Day 01 ~ Day ${String(totalDays).padStart(2, '0')})`;

      for (let day = 1; day <= totalDays; day++) {
        const readCount = localStorage.getItem(`n1_read_count_${day}`) || 0;
        const countInThisDay = Math.min(WORDS_PER_DAY, totalWords - (day - 1) * WORDS_PER_DAY);
        
        const dayCard = document.createElement('div');
        dayCard.className = 'day-card';
        dayCard.onclick = () => startStudy(day);
        dayCard.innerHTML = `
          <div>
            <div class="day-title">DAY ${String(day).padStart(2, '0')}</div>
            <span class="read-count">${readCount}회독 완료</span>
          </div>
          <div style="font-size:0.8rem; color:var(--text-sub); display:flex; justify-content:space-between;">
            <span>${countInThisDay} 단어</span>
            <span>학습하기 →</span>
          </div>
        `;
        grid.appendChild(dayCard);
      }

      document.getElementById('day-list-view').style.display = 'block';
      document.getElementById('study-view').style.display = 'none';
      document.getElementById('summary-view').style.display = 'none';
    }

    // 학습 시작
    function startStudy(day) {
      currentDay = day;
      isReviewOnlyMode = false;
      currentCards = getDayWords(day);
      if (currentCards.length === 0) return;
      currentIndex = 0;

      document.getElementById('day-list-view').style.display = 'none';
      document.getElementById('summary-view').style.display = 'none';
      document.getElementById('study-view').style.display = 'flex';
      showCard();
    }

    // 카드 출력
    function showCard() {
      isFlipped = false;
      document.getElementById('flashcard').classList.remove('flipped');
      const card = currentCards[currentIndex];
      document.getElementById('card-kanji').innerText = card.kanji;
      document.getElementById('card-hiragana').innerText = card.hiragana;
      document.getElementById('card-meaning').innerText = card.meaning;
      document.getElementById('study-progress-text').innerText = `${currentIndex + 1} / ${currentCards.length}`;
      document.getElementById('progress-bar').style.width = `${((currentIndex + 1) / currentCards.length) * 100}%`;
    }

    function flipCard() {
      isFlipped = !isFlipped;
      document.getElementById('flashcard').classList.toggle('flipped', isFlipped);
    }

    function markCard(type) {
      localStorage.setItem(`status_${currentCards[currentIndex].id}`, type);
      nextCard();
    }

    function nextCard() {
      if (currentIndex < currentCards.length - 1) {
        currentIndex++;
        showCard();
      } else {
        finishDay();
      }
    }

    function prevCard() {
      if (currentIndex > 0) {
        currentIndex--;
        showCard();
      }
    }

    // 완료 화면
    function finishDay() {
      if (!isReviewOnlyMode) {
        const curr = parseInt(localStorage.getItem(`n1_read_count_${currentDay}`) || 0);
        localStorage.setItem(`n1_read_count_${currentDay}`, curr + 1);
      }

      let memorized = 0, review = 0;
      currentCards.forEach(c => {
        const st = localStorage.getItem(`status_${c.id}`);
        if (st === 'memorized') memorized++;
        else if (st === 'review') review++;
      });

      document.getElementById('stat-memorized').innerText = memorized;
      document.getElementById('stat-review').innerText = review;
      document.getElementById('review-count-badge').innerText = review;
      document.getElementById('btn-retry-review').style.display = review > 0 ? 'block' : 'none';

      document.getElementById('study-view').style.display = 'none';
      document.getElementById('summary-view').style.display = 'flex';
    }

    function startReviewOnly() {
      isReviewOnlyMode = true;
      currentCards = currentCards.filter(c => localStorage.getItem(`status_${c.id}`) === 'review');
      if (currentCards.length === 0) return;
      currentIndex = 0;
      document.getElementById('summary-view').style.display = 'none';
      document.getElementById('study-view').style.display = 'flex';
      showCard();
    }

    function restartEntireDay() { startStudy(currentDay); }
    function goBackToDays() { renderDayList(); }

    // JSON 업로드 로직 (3000개 단어 처리)
    function importVocabulary(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          const data = JSON.parse(e.target.result);
          if (!Array.isArray(data)) {
            alert('JSON 파일 형태가 올바른 배열([ ]) 형식이 아닙니다.');
            return;
          }
          localStorage.setItem('N1_VOCAB_DATA', JSON.stringify(data));
          alert(`N1 단어 ${data.length}개가 성공적으로 불러와졌습니다!`);
          renderDayList();
        } catch (err) {
          alert('JSON 파일 형식이 올바르지 않습니다.');
        }
      };
      reader.readAsText(file);
    }

    // 키보드 단축키
    document.addEventListener('keydown', (e) => {
      const studyView = document.getElementById('study-view');
      if (studyView.style.display === 'flex') {
        if (e.key === ' ') { e.preventDefault(); flipCard(); }
        else if (e.key === 'ArrowRight') nextCard();
        else if (e.key === 'ArrowLeft') prevCard();
        else if (e.key === '1') markCard('review');
        else if (e.key === '2') markCard('memorized');
      }
    });

    window.onload = renderDayList;
  </script>
